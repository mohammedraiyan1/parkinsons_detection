# ──────────────────────────────────────────────
# SETUP INSTRUCTIONS
# ──────────────────────────────────────────────
#
# 1. Backend setup
# ─────────────────
#   cd backend
#   python -m venv venv
#   source venv/bin/activate        # Windows: venv\Scripts\activate
#   pip install -r requirements.txt
#
#   mkdir -p data models
#
# 2. Download UCI Parkinson's dataset
# ────────────────────────────────────
#   Option A (direct):
#     wget https://archive.ics.uci.edu/ml/machine-learning-databases/parkinsons/parkinsons.data
#     mv parkinsons.data data/
#
#   Option B (Kaggle):
#     kaggle datasets download -d nidaguler/parkinsons-data-set
#     unzip -j parkinsons-data-set.zip "*.data" -d data/
#
# 3. Train the model
# ───────────────────
#   python train_model.py
#   # → saves models/parkinsons_xgb.pkl
#   # → saves plots/model_evaluation.png
#   # → prints accuracy, AUC, SHAP importances
#
# 4. Start the backend
# ─────────────────────
#   python app.py
#   # → running on http://localhost:5000
#   # → test: curl http://localhost:5000/api/health
#
# 5. Frontend setup
# ──────────────────
#   cd ../frontend
#   npm install
#   echo "VITE_API_URL=http://localhost:5000" > .env
#   npm run dev
#   # → running on http://localhost:5173
#
# 6. Frontend dependencies needed
# ────────────────────────────────
#   npm install axios recharts framer-motion
#
# 7. Deploy (Render.com — free tier)
# ────────────────────────────────────
#   a. Push to GitHub
#   b. render.com → New Web Service → connect repo → backend folder
#   c. Build command:  pip install -r requirements.txt
#   d. Start command:  gunicorn app:app
#   e. Add env var:    PYTHON_VERSION=3.11.0
#
# ──────────────────────────────────────────────
